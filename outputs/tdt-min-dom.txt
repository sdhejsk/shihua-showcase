const token = '04265e698b77d4fd1d990d5e69d65647';
const map = L.map('map').setView([-25, 135], 4);
const base = L.tileLayer(`http://t{s}.tianditu.gov.cn/img_w/wmts?SERVICE=WMTS&REQUEST=GetTile&VERSION=1.0.0&LAYER=img&STYLE=default&TILEMATRIXSET=w&FORMAT=tiles&TILEMATRIX={z}&TILEROW={y}&TILECOL={x}&tk=${token}`, { subdomains:['0','1','2','3','4','5','6','7'], maxZoom:18, attribution:'tdt' });
base.on('tileload', () => document.body.setAttribute('data-tdt','loaded'));
base.on('tileerror', () => document.body.setAttribute('data-tdt','error'));
base.addTo(map);
setTimeout(() => map.invalidateSize(), 200);
</script>
